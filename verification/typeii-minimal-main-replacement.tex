\begin{targettheorem}[固定无理速率下的径向单泡 II 型爆破]\label{thm:main}
考虑三维聚焦能量临界 Hartree 方程
\[
 i\partial_tu+\Delta u+
 \kappa_{\rm H}(|x|^{-2}*|u|^4)|u|^2u=0,
 \qquad
 W(x)=(1+|x|^2)^{-1/2},
\]
其中常数 $\kappa_{\rm H}$ 取使 $W$ 满足相应定态方程的规范化。
固定一个无理速度
\[
 \nu_*=\sqrt2,\qquad
 \mu_*=\frac{\sqrt2}{2}>\frac9{16},\qquad
 \vartheta_*=0.
\]
则存在 $T>0$、一个径向强解
$u\in C([0,T);\dot H^1_{\rm rad}(\mathbb R^3))$、
正连续尺度 $\lambda(t)$、实相位 $\gamma(t)$、
$\eta_T\in\dot H^1_{\rm rad}(\mathbb R^3)$ 和
$\eta\in C([0,T);\dot H^1_{\rm rad}(\mathbb R^3))$，
使 $u$ 在 $[0,T)$ 上满足上述真实 Hartree 方程，并且
\[
 u(t,x)=e^{i\gamma(t)}\lambda(t)^{1/2}W(\lambda(t)x)
          +\eta(t,x),
 \qquad
 \eta(t)\longrightarrow\eta_T
 \quad\text{于 }\dot H^1\quad (t\uparrow T),
\]
\[
 \lambda(t)^{-1}
   =C_\lambda(T-t)^{\frac12+\sqrt2}(1+o(1)),
 \qquad C_\lambda>0.
\]
解具有有限时 II 型爆破性质：
\[
 \sup_{0\le t<T}\|u(t)\|_{\dot H^1}<\infty,
 \qquad
 T<\infty,
\]
且 $T$ 是该强解最大前向定义区间的右端点，即
$u(t)$ 在 $t\uparrow T$ 时不存在 $\dot H^1$ 强极限。
\end{targettheorem}
