\begin{targettheorem}[Hartree 动力学选速的径向单泡 II 型爆破]
\label{thm:main}
考虑三维聚焦能量临界 Hartree 方程
\[
 i\partial_tu+\Delta u+
 \kappa_{\rm H}(|x|^{-2}*|u|^4)|u|^2u=0,
 \qquad
 W(x)=(1+|x|^2)^{-1/2},
\]
其中 $\kappa_{\rm H}>0$ 使 $-\Delta W=\mathcal N(W)$。
则存在有限时间 $T>t_0$、一个真实径向强解
$u\in C([t_0,T);\dot H^1_{\rm rad}(\mathbb R^3))$、
由该解的两通道调制正交条件确定的 $C^1$ 尺度与相位
$\lambda:[t_0,T)\to(0,\infty)$、
$\gamma:[t_0,T)\to\mathbb R$，
以及 $\eta_T\in\dot H^1_{\rm rad}(\mathbb R^3)$，使
\[
 u(t,x)=e^{i\gamma(t)}\lambda(t)^{1/2}W(\lambda(t)x)
   +\eta(t,x),
 \qquad
 \eta(t)\longrightarrow\eta_T
 \quad\text{于 }\dot H^1\quad (t\uparrow T).
\]
定义实际重整化时钟、缩放与调制坐标
\[
 \frac{\mathrm ds}{\mathrm dt}=\lambda(t)^2,
 \quad s(t)\longrightarrow+\infty,\qquad
 b(s)=\frac{\lambda_s}{\lambda},\qquad
 \mathfrak q(s)=(T-t(s))\lambda(t(s))^2,
 \qquad
 \mu(s)=\frac{2b(s)\mathfrak q(s)-1}{4}.
\]
则存在 $\delta\in(0,1)$、$C>0$ 及一个由此真实动力学产生的极限
$\mu_\infty>9/16$，使
\[
 \lim_{s\to\infty}\mu(s)=\mu_\infty,
 \qquad |\mu_s(s)|\le Cs^{-1-\delta},
 \qquad |\mu(s)-\mu_\infty|\le\frac C\delta s^{-\delta}
 \quad(s\gg1).
\]
因此有一个由实际解确定、而非预先指定的集中指数
\[
 a_\infty(u_0):=\frac12+2\mu_\infty>\frac12,
 \qquad
 \lambda(t)^{-1}
 =C_\lambda(u_0)(T-t)^{a_\infty(u_0)}(1+o(1)),
 \qquad C_\lambda(u_0)>0.
\]
该解具有真正的有限时径向 II 型爆破：
\[
 T<\infty,\qquad
 \sup_{t_0\le t<T}\|u(t)\|_{\dot H^1}<\infty,
\]
并且不能在 $\dot H^1$ 中强延拓穿过 $T$；
等价地，在该解的最大前向强解区间中，$T$ 是右端点。
\end{targettheorem}
